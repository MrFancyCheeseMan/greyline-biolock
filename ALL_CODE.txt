-- -------------------------------------------------------------------------
-- Accessors -- always go through convars so runtime changes take effect.
-- -------------------------------------------------------------------------
local function cv(name, fallback)
    local c = GetConVar(name)
    return c and c:GetString() or fallback
end

local function cvFloat(name, fallback)
    local c = GetConVar(name)
    return tonumber(c and c:GetString() or nil) or fallback
end

local function bool(value)
    return value == "1" or string.lower(value) == "true" or string.lower(value) == "yes"
end

local function cvVector(name, fallback)
    local raw = cv(name, "")
    local x, y, z = string.match(raw, "^%s*(-?[%d%.]+)[%s,]+(-?[%d%.]+)[%s,]+(-?[%d%.]+)%s*$")
    if not x then return fallback end
    return Vector(tonumber(x), tonumber(y), tonumber(z))
end

function Biolock.GetLightOffset()
    local offset = cvVector(Biolock.Convars.light_offset[1], Biolock.Config.LightOffset)
    -- Archived servers may still hold the old "+X is the front" value. The
    -- combine_lock01 face is +Y, so rotate that legacy offset into place.
    if math.abs(offset.y) < 0.01 and math.abs(offset.x) > 0.01 then
        offset = Vector(0, offset.x, offset.z)
    end
    return offset
end

-- combine_lock01's visible face is along local +Y, while an entity's forward
-- is local +X. A -90 yaw makes the modelled face point straight out of the
-- door. The convar lets an admin correct this live if a custom model differs.
function Biolock.GetModelAngleOffset()
    local yaw = math.Clamp(cvFloat(Biolock.Convars.model_yaw_offset[1],
        Biolock.Config.ModelYawOffset), -180, 180)
    return Angle(0, yaw, 0)
end

function Biolock.GetAllowedFactions()
    local raw = cv(Biolock.Convars.allowed_factions[1], "combine")
    local allowed = {}
    for part in string.gmatch(raw, "[^,]+") do
        part = string.lower(string.Trim(part))
        if part ~= "" then allowed[part] = true end
    end
    if not next(allowed) then allowed.combine = true end
    return allowed
end

-- -------------------------------------------------------------------------
-- GAMEMODE INTEGRATION
--
-- Greyline stores faction in this order (see sh_items.lua in the gamemode):
--   ply.GLFaction          temporary override set by an admin/event
--   ply.GLCharacter.faction persisted character record
--   gl_faction NWString    replicated to clients
-- Override this function if your gamemode exposes a different API.
-- -------------------------------------------------------------------------
function Biolock.GetPlayerFaction(ply)
    if not IsValid(ply) then return nil end

    if ply.GLFaction and ply.GLFaction ~= "" then
        return string.lower(tostring(ply.GLFaction))
    end
    if ply.GLCharacter and ply.GLCharacter.faction then
        return string.lower(tostring(ply.GLCharacter.faction))
    end

    local replicated = ply:GetNWString("gl_faction", "")
    if replicated and replicated ~= "" then
        return string.lower(replicated)
    end

    if isfunction(ply.GetFaction) then
        return string.lower(tostring(ply:GetFaction()))
    end

    return nil
end

function Biolock.HasAdminOverride(ply)
    local mode = Biolock.GetAdminOverrideMode()
    if mode == "none" or not IsValid(ply) then return false end
    if mode == "admin" then return ply:IsAdmin() end
    return ply:IsSuperAdmin()
end

function Biolock.CanUnlock(ply)
    if not IsValid(ply) or not Biolock.IsEnabled() then return false end
    if Biolock.HasAdminOverride(ply) then return true end

    local faction = Biolock.GetPlayerFaction(ply)
    return faction ~= nil and Biolock.GetAllowedFactions()[faction] == true
end

function Biolock.CanPlace(ply)
    if not IsValid(ply) or not Biolock.IsEnabled() then return false end
    if Biolock.HasAdminOverride(ply) then return true end
    return Biolock.AllowFactionPlace() and Biolock.CanUnlock(ply)
end

function Biolock.IsDoor(ent)
    return IsValid(ent) and Biolock.Config.DoorClasses[ent:GetClass()] == true
end

-- Return the signed min/max extent of the door along a world-space axis,
-- relative to its centre. Used to find the handle edge, not the door centre.
local function extentsAlong(door, center, axis)
    local mins, maxs = door:GetCollisionBounds()
    if not mins or not maxs then
        mins, maxs = door:OBBMins(), door:OBBMaxs()
    end
    if not mins or not maxs then return -24, 24 end

    local low, high = math.huge, -math.huge
    for _, x in ipairs({ mins.x, maxs.x }) do
        for _, y in ipairs({ mins.y, maxs.y }) do
            for _, z in ipairs({ mins.z, maxs.z }) do
                local along = (door:LocalToWorld(Vector(x, y, z)) - center):Dot(axis)
                low = math.min(low, along)
                high = math.max(high, along)
            end
        end
    end
    if low > high then return -24, 24 end
    return low, high
end

-- The lock is ONLY mounted at the handle edge of the door. For hinged doors
-- the entity origin is the hinge, so the handle is the opposite edge. Brush
-- doors use the side the player aimed at. Interactive placement must be aimed
-- within HandleBand of that edge; restore calls (no player) skip the check.
function Biolock.GetPlacement(door, pos, ang, ply)
    if not IsValid(door) then return pos or vector_origin, ang or angle_zero end

    local center = door:WorldSpaceCenter()
    local outward = (ang and ang:Forward()) or Vector(0, 0, 0)
    if outward:LengthSqr() < 0.01 and pos then outward = pos - center end
    outward.z = 0
    if outward:LengthSqr() < 0.01 then outward = Vector(1, 0, 0) end
    outward:Normalize()

    local right = outward:Cross(Vector(0, 0, 1))
    if right:LengthSqr() < 0.01 then right = Vector(0, 1, 0) end
    right:Normalize()

    local low, high = extentsAlong(door, center, right)
    local heightLow, heightHigh = extentsAlong(door, center, Vector(0, 0, 1))
    local halfSpan = (high - low) * 0.5

    -- Hinge at the entity origin => handle is the opposite edge.
    local originAlong = (door:GetPos() - center):Dot(right)
    local handleSide
    if math.abs(originAlong) > halfSpan * 0.5 then
        handleSide = originAlong > 0 and -1 or 1
    elseif pos then
        handleSide = (pos - center):Dot(right) >= 0 and 1 or -1
    else
        handleSide = 1
    end

    local edge = handleSide > 0 and high or low
    local inset = math.Clamp(isfunction(Biolock.GetHandleInset)
        and Biolock.GetHandleInset() or Biolock.Config.HandleInset, 2, math.max(2, halfSpan - 2))
    local along = edge - handleSide * inset

    -- Handle height comes from the trace, clamped to the physical door.
    local z = center.z
    if pos then
        z = math.Clamp(pos.z, center.z + heightLow + 3, center.z + heightHigh - 3)
    end

    if pos and IsValid(ply) then
        local band = isfunction(Biolock.GetHandleBand)
            and Biolock.GetHandleBand() or Biolock.Config.HandleBand
        local aimAlong = (pos - center):Dot(right)
        if math.abs(aimAlong - along) > band or math.abs(pos.z - z) > band then
            return nil, nil -- not aimed at the handle
        end
    end

    local _, _, outOffset = Biolock.GetHandleOffset()
    local place = center + right * along + Vector(0, 0, z - center.z) + outward * outOffset
    local angleOffset = isfunction(Biolock.GetModelAngleOffset)
        and Biolock.GetModelAngleOffset()
        or (isangle(Biolock.Config.ModelAngleOffset) and Biolock.Config.ModelAngleOffset or angle_zero)
    local facing = Angle(0, outward:Angle().yaw, 0) + angleOffset

    return place, facing
end

local Biolock = Greyline.Biolock
local STATE  = Biolock.Config.State
local PERSIST_FILE = "greyline_biolocks.json"

-- door EntIndex -> biolock entity. Rebuilt on map change.
Biolock.ByDoor = Biolock.ByDoor or {}

-- A double door is two separate door entities. One scanner covers both leaves;
-- the bare leaf keeps only a back-reference to the real lock.
local function modelYaw()
    if isfunction(Biolock.GetModelAngleOffset) then return Biolock.GetModelAngleOffset().y end
    return isangle(Biolock.Config.ModelAngleOffset) and Biolock.Config.ModelAngleOffset.y or -90
end

local function outwardFromFacing(facing)
    return Angle(0, (facing and facing.y or 0) - modelYaw(), 0):Forward()
end

local function sameDoorName(a, b)
    local name = a:GetName()
    return name ~= nil and name ~= "" and name == b:GetName()
end

function Biolock.GetPairedDoor(door, outwardHint)
    if not IsValid(door) or not Biolock.IsDoor(door) then return nil end

    if IsValid(door.GLBiolockPair) and Biolock.IsDoor(door.GLBiolockPair) then
        return door.GLBiolockPair
    end

    local range = isfunction(Biolock.GetDoubleDoorRange) and Biolock.GetDoubleDoorRange() or 96
    if range <= 0 then return nil end

    local center = door:WorldSpaceCenter()
    local outward = outwardHint or door:GetAngles():Forward()
    outward.z = 0
    if outward:LengthSqr() < 0.01 then outward = Vector(1, 0, 0) end
    outward:Normalize()

    local best, bestScore = nil, math.huge
    for _, other in ipairs(ents.GetAll()) do
        if other ~= door and IsValid(other) and Biolock.IsDoor(other) then
            local delta = other:WorldSpaceCenter() - center
            local lateral = delta - outward * delta:Dot(outward)
            local span = lateral:Length()
            local named = sameDoorName(door, other)
            local maxSpan = named and 256 or range

            if math.abs(delta.z) <= 48 and math.abs(delta:Dot(outward)) <= 24
                and span <= maxSpan then
                local score = named and (span - 10000) or span
                if score < bestScore then best, bestScore = other, score end
            end
        end
    end

    if IsValid(best) then
        door.GLBiolockPair = best
        best.GLBiolockPair = door
    end
    return best
end

function Biolock.Register(lock, door)
    Biolock.ByDoor[door:EntIndex()] = lock
end

function Biolock.Unregister(lock)
    for index, candidate in pairs(Biolock.ByDoor) do
        if candidate == lock then Biolock.ByDoor[index] = nil end
    end
end

function Biolock.GetLockForDoor(door)
    if not IsValid(door) then return nil end

    local lock = Biolock.ByDoor[door:EntIndex()]
    if IsValid(lock) and lock:GetDoor() == door then return lock end

    local pair = door.GLBiolockPair
    if IsValid(pair) then
        local pairedLock = Biolock.ByDoor[pair:EntIndex()]
        if IsValid(pairedLock) and pairedLock:GetDoor() == pair then return pairedLock end
    end

    return nil
end

function Biolock.IsLockedDoor(door)
    if not IsValid(door) then return false end
    local lock = Biolock.GetLockForDoor(door)
    return door.GLBiolockLocked == true or (IsValid(lock) and lock:GetLocked())
end

--[[
    Lock/unlock a door for every supported door class. On a double door the
    paired leaf receives the same state, so one scanner covers both leaves.

    Fire("Lock") is the canonical brush-door API. We additionally pin the
    movetype to NONE while locked: prop_door_rotating is a physics entity and
    can otherwise be shoved open with a crowbar, physgun or another prop.
    The original movetype is restored on unlock/removal.
]]
local function applyDoorState(door, locked)
    if locked then
        if door.GLBiolockOriginalMoveType == nil then
            door.GLBiolockOriginalMoveType = door:GetMoveType()
        end

        door:Fire("Lock", "", 0)
        door:SetMoveType(MOVETYPE_NONE)
        door.GLBiolockLocked = true
    else
        door:Fire("Unlock", "", 0)

        if door.GLBiolockOriginalMoveType ~= nil then
            door:SetMoveType(door.GLBiolockOriginalMoveType)
            door.GLBiolockOriginalMoveType = nil
        end

        door.GLBiolockLocked = nil
    end
end

function Biolock.SetDoorLocked(door, locked, lock)
    if not IsValid(door) then return end

    applyDoorState(door, locked)

    local pair = Biolock.GetPairedDoor(door, IsValid(lock) and (lock.GLOutward or lock:GetRight()) or nil)
    if IsValid(pair) and pair ~= door then applyDoorState(pair, locked) end

    if IsValid(lock) then lock.GLLocked = locked end
end

-- Create a scanner and attach it. pos/ang are optional hints; the final
-- handle-side placement is always resolved from the door on the server.
function Biolock.Create(door, pos, ang, ply)
    if not Biolock.IsDoor(door) then return nil end
    if Biolock.GetLockForDoor(door) then return nil end

    local place, facing = Biolock.GetPlacement(door, pos, ang, ply)
    if not place then
        if IsValid(ply) then
            Biolock.Feedback(ply, "denied", "AIM AT THE DOOR HANDLE", 0)
            Biolock.PlaySound(ply, nil, "denied")
        end
        return nil
    end

    -- Cache the adjacent leaf before the duplicate check so a second scanner
    -- cannot be placed on the other half of a double door.
    local outward = outwardFromFacing(facing)
    Biolock.GetPairedDoor(door, outward)
    if Biolock.GetLockForDoor(door) then
        if IsValid(ply) then
            Biolock.Feedback(ply, "denied", "LOCK ALREADY PRESENT", 0)
            Biolock.PlaySound(ply, nil, "denied")
        end
        return nil
    end

    local lock = ents.Create("greyline_biolock")
    if not IsValid(lock) then return nil end

    lock.GLOutward = outward
    lock:SetPos(place)
    lock:SetAngles(facing)
    lock:Spawn()
    lock:Activate()
    lock:AttachToDoor(door, ply)

    return lock
end

end)

-- -------------------------------------------------------------------------
-- PERSISTENCE -- optional, opt-in via convar
-- -------------------------------------------------------------------------
-- util.TableToJSON / JSONToTable round-trip Vectors and Angles as plain tables.
-- Accept both the short (x/y/z, p/y/r) and long (pitch/yaw/roll) key forms.
local function toVector(t)
    if not istable(t) then return Vector(0, 0, 0) end
    return Vector(tonumber(t.x) or 0, tonumber(t.y) or 0, tonumber(t.z) or 0)
end

local function toAngle(t)
    if not istable(t) then return angle_zero end
    local pitch = tonumber(t.p or t.pitch) or 0
    local yaw   = tonumber(t.y or t.yaw) or 0
    local roll  = tonumber(t.r or t.roll) or 0
    return Angle(pitch, yaw, roll)
end

function Biolock.SaveState()
    if not Biolock.GetPersist() then return end

    local entries = {}
    for _, lock in pairs(Biolock.ByDoor) do
        local door = IsValid(lock) and lock:GetDoor() or nil
        if IsValid(door) then
            entries[#entries + 1] = {
                scannerPos = lock:GetPos(),
                scannerAng = lock:GetAngles(),
                doorPos    = door:GetPos(),
                doorClass  = door:GetClass(),
            }
        end
    end

    file.Write(PERSIST_FILE, util.TableToJSON(entries, true))
end

hook.Add("InitPostEntity", "Greyline.Biolock.Restore", function()
    timer.Simple(1, function()
        if not Biolock.GetPersist() then return end

        local raw = file.Read(PERSIST_FILE, "DATA")
        if not raw or raw == "" then return end

        local entries = util.JSONToTable(raw)
        if not istable(entries) then return end

        for _, entry in ipairs(entries) do
            local target = toVector(entry.doorPos)
            local best, bestDistance = nil, math.huge

            for _, door in ipairs(ents.FindByClass(entry.doorClass or "func_door")) do
                local distance = door:GetPos():Distance(target)
                if distance < bestDistance then
                    best, bestDistance = door, distance
                end
            end

            -- 64u tolerance keeps a stale save from locking an unrelated door.
            if IsValid(best) and bestDistance <= 64 and not Biolock.GetLockForDoor(best) then
                -- Reuse the saved scanner transform so the restored lock goes
                -- back to the same handle edge and height, not a guessed side.
                local savedPos = toVector(entry.scannerPos)
                local savedAng = toAngle(entry.scannerAng)
                local yawOffset = isfunction(Biolock.GetModelAngleOffset)
                    and Biolock.GetModelAngleOffset().y
                    or (isangle(Biolock.Config.ModelAngleOffset) and Biolock.Config.ModelAngleOffset.y or -90)
                local outwardAng = Angle(0, savedAng.y - yawOffset, 0)
                Biolock.Create(best, savedPos, outwardAng, nil)
            end
        end

        Biolock.Log("restored", nil, "locks=" .. tostring(#entries))
    end)
end)

hook.Add("Shutdown", "Greyline.Biolock.SaveOnShutdown", function()
    Biolock.SaveState()
end)

-- `give` is sandbox-only; provide an explicit way to obtain a lock. If the
-- gamemode exposes a server-authoritative inventory service, put the lock in
-- the inventory. Otherwise fall back to the standalone SWEP.
local function giveViaInventory(ply)
    local inventory = Greyline and Greyline.Inventory
    if not inventory or not isfunction(inventory.Provision) then return false end

    local ok, reason = inventory.Provision(ply, "biolock", 1)
    if ok then
        ply:ChatPrint("[GREYLINE] Biolock added to your inventory. Aim at a door, open it, then left-click the item.")
        if isfunction(Biolock.Feedback) then
            Biolock.Feedback(ply, "granted", "BIOLOCK ADDED", 1.5)
        end
    else
        ply:ChatPrint("[GREYLINE] " .. tostring(reason))
    end
    return true
end

===== lua/entities/greyline_biolock/init.lua =====
--[[---------------------------------------------------------------------------
    Greyline :: Biolock entity -- server
    File: lua/entities/greyline_biolock/init.lua

    Owns the scan state machine. Permission checks happen in Use(); the timer
    completion is re-validated in Think() so a disconnect/teleport cancels it.
---------------------------------------------------------------------------]]

AddCSLuaFile("cl_init.lua")
AddCSLuaFile("shared.lua")
include("shared.lua")

local Biolock = Greyline.Biolock
local STATE  = Biolock.Config.State

function ENT:Initialize()
    local model = Biolock.GetModel()
    self:SetModel(model)

    -- No physics object: physgun/gravity gun cannot grab or nudge the device.
    self:SetMoveType(MOVETYPE_NONE)
    self:SetSolid(SOLID_BBOX)
    self:SetSize(Vector(-6, -6, -2), Vector(6, 6, 10))
    self:SetUseType(SIMPLE_USE)

    self.GLLocked    = false
    self.GLHealth    = 100
    self.GLScanningBy = nil
    self.GLScanEndsAt = 0

    self:SetNWInt("greyline_biolock_state", STATE.Locked)
    self:SetNWFloat("greyline_biolock_scan_start", 0)
    self:SetNWFloat("greyline_biolock_scan_end", 0)
end

-- Called by Biolock.Create / the placement SWEP immediately after spawning.
function ENT:AttachToDoor(door, ply)
    if not Biolock.IsDoor(door) then return false end
    if Biolock.GetLockForDoor(door) then return false end

    self.GLDoor = door
    door.GLBiolock = self

    -- Ride the door: parenting makes the scanner follow every open/close,
    -- including rotating doors. SetParent keeps the child's local transform,
    -- so reapply the world placement chosen by Biolock.Create afterwards.
    local worldPos, worldAng = self:GetPos(), self:GetAngles()
    self:SetParent(door)
    self:SetPos(worldPos)
    self:SetAngles(worldAng)

    Biolock.Register(self, door)
    Biolock.SetDoorLocked(door, true, self)

    self.GLLocked = true
    self:SetNWInt("greyline_biolock_state", STATE.Locked)

    Biolock.Log("placed", ply, string.format("door=%d class=%s",
        door:EntIndex(), door:GetClass()))
    Biolock.SaveState()

    return true
end

function ENT:Use(activator, caller)
    if not IsValid(activator) or not activator:IsPlayer() then return end
    if not Biolock.IsEnabled() then return end

    local door = self:GetDoor()
    if not IsValid(door) then
        self:Remove()
        return
    end

    local current = self:GetState()
    if current == STATE.Scanning then return end

    -- Per-device debounce: SIMPLE_USE can fire several times per press.
    if self.GLNextUse and CurTime() < self.GLNextUse then return end
    self.GLNextUse = CurTime() + 0.5

    local function deny()
        self:SetNWInt("greyline_biolock_state", STATE.Denied)
        self.GLDeniedUntil = CurTime() + 1

        Biolock.Feedback(activator, "denied", "ACCESS DENIED", 0)
        Biolock.PlaySound(activator, self:GetPos(), "denied")
        Biolock.Log("access_denied", activator, "door=" .. tostring(door:EntIndex()))
    end

    -- THE permission check. Everything else here is cosmetic.
    if not Biolock.CanUnlock(activator) then
        deny()
        return
    end

    -- Toggle: once granted, the door stays unlocked until an authorized
    -- player clicks the scanner again. Unauthorized clicks still deny only.
    if current == STATE.Granted then
        Biolock.SetDoorLocked(door, true, self)

        self.GLScanningBy = nil
        self:SetNWInt("greyline_biolock_state", STATE.Locked)
        self:SetNWFloat("greyline_biolock_scan_start", 0)
        self:SetNWFloat("greyline_biolock_scan_end", 0)

        Biolock.Feedback(activator, "locked", "DOOR LOCKED", 0)
        Biolock.PlaySound(activator, self:GetPos(), "granted")
        Biolock.Log("relocked", activator, "door=" .. tostring(door:EntIndex()))
        Biolock.SaveState()
        return
    end

    local duration = Biolock.GetScanDuration()
    local now = CurTime()

    self.GLScanningBy  = activator
    self.GLScanEndsAt  = now + duration

    self:SetNWInt("greyline_biolock_state", STATE.Scanning)
    self:SetNWFloat("greyline_biolock_scan_start", now)
    self:SetNWFloat("greyline_biolock_scan_end", self.GLScanEndsAt)

    Biolock.Feedback(activator, "scanning", "BIOMETRIC SCAN", duration)
    Biolock.PlaySound(activator, self:GetPos(), "scan")
end

function ENT:Think()
    local now = CurTime()

    -- Cheap liveness check. If the door was removed, the device goes with it.
    if now >= (self.GLNextCheck or 0) then
        self.GLNextCheck = now + 0.5
        if not IsValid(self:GetDoor()) then
            self:Remove()
            return
        end
    end

    local state = self:GetState()

    if state == STATE.Scanning then
        local ply = self.GLScanningBy

        -- Disconnect, death or walking away all abort the scan.
        if not IsValid(ply) or not ply:Alive()
            or ply:GetPos():Distance(self:GetPos()) > Biolock.GetUseDistance() then
            self.GLScanningBy = nil
            self:SetNWInt("greyline_biolock_state", STATE.Locked)
            self:SetNWFloat("greyline_biolock_scan_start", 0)
            self:SetNWFloat("greyline_biolock_scan_end", 0)
            return
        end

        if now >= (self.GLScanEndsAt or math.huge) then
            self.GLScanningBy = nil

            local door = self:GetDoor()
            if IsValid(door) then
                Biolock.SetDoorLocked(door, false, self)
            end

            self:SetNWInt("greyline_biolock_state", STATE.Granted)
            self:SetNWFloat("greyline_biolock_scan_end", 0)
            -- No timer: Granted is latched until the next authorized Use.

            Biolock.Feedback(ply, "granted", "ACCESS GRANTED", 0)
            Biolock.PlaySound(ply, self:GetPos(), "granted")
            Biolock.Log("access_granted", ply, "door=" .. tostring(door:EntIndex()))
            Biolock.SaveState()
        end

        return
    end

    if state == STATE.Granted then
        -- Intentionally no auto-relock. ENT:Use handles the next click.
        return
    end

    if state == STATE.Denied and now >= (self.GLDeniedUntil or 0) then
        -- A denied click must not erase a latched unlock. Restore the actual
        -- door state rather than blindly returning to Locked.
        local restored = self:GetLocked() and STATE.Locked or STATE.Granted
        self:SetNWInt("greyline_biolock_state", restored)
    end
end

function ENT:OnRemove()
    local door = self:GetDoor()

    if IsValid(door) and self:GetLocked() then
        Biolock.SetDoorLocked(door, false, self) -- never strand a locked door
    end
    if IsValid(door) then door.GLBiolock = nil end

    Biolock.Unregister(self)
    Biolock.SaveState()
end

function ENT:OnTakeDamage(dmginfo)
    if not Biolock.IsDestroyable() then return 0 end

    self.GLHealth = (self.GLHealth or 100) - dmginfo:GetDamage()
    if self.GLHealth <= 0 then
        local attacker = dmginfo:GetAttacker()
        Biolock.Log("destroyed", IsValid(attacker) and attacker:IsPlayer() and attacker or nil,
            "door=" .. tostring(IsValid(self:GetDoor()) and self:GetDoor():EntIndex() or -1))
        self:Remove() -- OnRemove unlocks the door
    end

    return 0
end

    if state == STATE.Granted then
        colour = palette.granted
    elseif state == STATE.Scanning then
        colour = palette.scanning
    elseif state == STATE.Denied then
        colour = palette.denied
    else
        colour = palette.locked
    end

        and (0.65 + 0.35 * math.sin(CurTime() * 9)) or 1

        or Vector(0, 8, -6)

    -- Scan progress: a sprite that expands across the duration. All values
    -- come from the server's start/end timestamps, so it cannot be faked.
    if state == STATE.Scanning then
        local started = self:GetScanStart()
        local finished = self:GetScanEnd()
        local duration = math.max(finished - started, 0.001)
        local frac = math.Clamp((CurTime() - started) / duration, 0, 1)

        render.DrawSprite(pos + self:GetForward() * 2,
            4 + 20 * frac, 4 + 20 * frac,
            Color(colour.r, colour.g, colour.b, 150))
    end

# Greyline Biolock

Combine-style biometric door lock for a custom Garry's Mod gamemode. The
scanner uses the gamemode's existing faction system; only Combine players
(plus an admin override) can unlock a door it is attached to.

## Install

1. Copy this folder to `garrysmod/addons/greyline-biolock/`.
2. Restart the server (autorun files only load at startup).
3. Obtain a lock with `greyline_biolock_give` (the engine's `give` command only
   exists in sandbox). If the gamemode exposes `Greyline.Inventory.Provision`,
   the lock is added as an inventory item; otherwise the standalone SWEP is
   given. `greyline_biolock_place` remains a console fallback while aiming at a
   door.
4. With the inventory item, aim at a door, open your belongings, hover the
   Biolock (`Use`) and left-click it. With the SWEP, primary fire attaches it.
5. Remove a lock with the tool's secondary fire or `greyline_biolock_remove`.

No workshop content is required. The default biolock model
(`models/props_combine/combine_lock01.mdl`) and sounds ship with HL2/GMod; all
paths are configurable via convars.

## Faction integration

`Biolock.GetPlayerFaction(ply)` in `lua/autorun/greyline_biolock_config.lua`
reads, in order:

1. `ply.GLFaction` (temporary override)
2. `ply.GLCharacter.faction` (persisted character)
3. `ply:GetNWString("gl_faction")`
4. `ply:GetFaction()` if present

If your gamemode uses a different API, override that one function. Do not
add a second faction system.

> **Greyline caveat:** the current showcase code's `normalizeCharacterFaction()`
> unconditionally resets `faction` to `civilian`. Until that is fixed, no player
> will report as Combine and every scan will be denied. Set `ply.GLFaction` to
> `"combine"` when testing, or fix the helper first.

## How it works

1. The tool traces from the server-side player eye to a door-class entity. The
   aim point must be near the handle edge; aiming at the door centre is rejected.
2. `Biolock.Create` snaps the scanner to the handle edge, parents it to the door
   (so it follows
   every open/close), calls `Fire("Lock")`, and pins the door's movetype to
   `MOVETYPE_NONE` so it cannot be physgunned or pushed open.
3. `PlayerUse`, `PhysgunPickup` and `GravGunPickupAllowed` hooks reject the
   locked door for everyone. `EntityTakeDamage` blocks damage unless the device
   is configured destroyable.
4. `ENT:Use` performs **the only** permission check, serverside. Denied players
   get a red HUD banner and beep; allowed players get a 1.2s blue pulse.
5. On success the door is unlocked and the scanner turns green. It stays
   unlocked until an authorized player clicks the scanner again to re-lock it;
   there is no automatic re-lock timer.
6. `OnRemove` always unlocks the door first, including when the scanner is
   destroyed, the door is removed, or the map changes.
7. **Double doors:** one scanner covers both leaves. The server pairs adjacent
   doors (same targetname, or nearest same-plane leaf within
   `greyline_biolock_double_door_range`, default 96). Lock/unlock affects both
   sides; a second scanner on the bare leaf is rejected. Range `0` disables it.

**No client-to-server net message exists.** The only net string is
server-to-client feedback.

## Configuration

All tunables are in `lua/autorun/greyline_biolock_config.lua`; convars use the
`greyline_biolock_` prefix. See the `Biolock.Convars` table for names/defaults.
`handle_inset` (10) and `handle_band` (18) snap the lock to the handle edge and
control how close you must aim; `handle_right`/`handle_down` are deprecated.
`handle_out` (4) lifts it off the door face. `model_yaw_offset` (-90) turns `combine_lock01`'s +Y face straight out of the door
(pitch/roll stay zero); `light_offset` (`0 8 -6`) moves the status glow onto the
model's circular light. `double_door_range` (96) pairs the two leaves of a double
door; set it to `0` to disable. `relock_delay` is deprecated and ignored.

## Testing checklist

Use this checklist:

- Non-Combine: Use the scanner -> red "ACCESS DENIED", door stays locked.
- Combine: Use the scanner -> blue progress -> green "ACCESS GRANTED", door opens.
- Superadmin: same as Combine even with faction set to civilian.
- Try `+use`, physgun, gravity gun and crowbar on the locked door -> all blocked.
- Remove the scanner while locked -> door unlocks.
- Disconnect mid-scan -> scan cancels, no unlock.
- Place on a wall -> "INVALID SURFACE". Aim at door centre -> "AIM AT THE DOOR
  HANDLE". Place twice -> "LOCK ALREADY PRESENT".
- Set `greyline_biolock_persist 1`, change map -> locks restore.
